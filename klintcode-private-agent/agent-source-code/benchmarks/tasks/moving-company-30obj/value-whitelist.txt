# TeaQL KSML Types are inferred from values!
# Normal fields: Use representative literal values.
# Examples:
# String: name="Fast Movers"
# Int: founded="1998"
# Long: external_id="1000000000000000000l"
# Double/Money: budget="50000.00"

# Special functions:
# create_time="createTime()"
# update_time="updateTime()"

# Relationships:
# Use object_name() to reference another object.
# Example: company="company()" refers to the <company> object.
